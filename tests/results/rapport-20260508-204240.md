enum RiskLevel: int {
    // Les valeurs numériques ici ne sont qu’indicatives.
    case LOW = 0;
    case MEDIUM = 300;
    case HIGH = 700;
    case CRITICAL = 901;

    public static function fromScore(int $score): self {
        if ($score >= 0 && $score <= 300) {
            return self::LOW;
        } elseif ($score >= 301 && $score <= 700) {
            return self::MEDIUM;
        } elseif ($score >= 701 && $score <= 900) {
            return self::HIGH;
        } elseif ($score >= 901 && $score <= 1000) {
            return self::CRITICAL;
        }
        throw new \InvalidArgumentException("Risk score out of valid range (0-1000): " . $score);
    }
}

class Transaction {
    public readonly string $id; // identifiant (ex: UUID)
    private Money $money;
    private string $clientId;
    private bool $isAccountSuspended; // invariant : compte non suspendu

    // Statuts possibles
    private const STATUS_PENDING = 'PENDING';
    private const STATUS_APPROVED = 'APPROVED';
    private const STATUS_REVIEW_REQUIRED = 'REVIEW_REQUIRED';
    private const STATUS_BLOCKED = 'BLOCKED';

    private string $status;
    /** @var object[] */
    private array $recordedEvents = [];

    public function __construct(string $id, string $clientId, Money $money, bool $isAccountSuspended = false) {
        $this->id = $id;
        $this->clientId = $clientId;
        $this->money = $money;
        $this->isAccountSuspended = $isAccountSuspended;
        $this->status = self::STATUS_PENDING;
    }

    /**
     * Accesseur pour l'argent (nécessaire pour la stratégie de scoring)
     */
    public function getMoney(): Money {
        return $this->money;
    }

    /**
     * Méthode d'invariant : soumettre la transaction.
     * Peut seulement être appelée en état PENDING.
     */
    public function submit(): void {
        if ($this->status !== self::STATUS_PENDING) {
            throw new \Exception("Transaction is not in PENDING state.");
        }
        // Vérifie l'invariant : compte non suspendu.
        if ($this->isAccountSuspended) {
            throw new \Exception("Cannot submit transaction; associated account is suspended.");
        }
        $this->recordEvent(new TransactionSubmittedEvent($this));
    }

    /**
     * Applique le score de risque et effectue la transition d'état.
     * Seule une transition depuis PENDING est autorisée.
     */
    public function applyRiskScore(RiskScore $riskScore): void {
        if ($this->status !== self::STATUS_PENDING) {
            throw new \Exception("Cannot apply risk score when not in PENDING state.");
        }
        // Décision de transition selon le niveau de risque :
        switch ($riskScore->level()) {
            case RiskLevel::CRITICAL:
                $this->status = self::STATUS_BLOCKED;
                $this->recordEvent(new FraudAlertTriggeredEvent($this));
                break;
            case RiskLevel::HIGH:
                $this->status = self::STATUS_REVIEW_REQUIRED;
                break;
            default: // LOW et MEDIUM
                $this->status = self::STATUS_APPROVED;
                break;
        }
        $this->recordEvent(new RiskScoreCalculatedEvent($this, $riskScore));
    }

    public function getStatus(): string {
        return $this->status;
    }

    /**
     * Retourne les événements enregistrés.
     */
    public function getRecordedEvents(): array {
        return $this->recordedEvents;
    }

    private function recordEvent(object $event): void {
        $this->recordedEvents[] = $event;
    }

                // Pour la démo, on utilise le transactionId fourni comme identifiant UUID.
                $money = new Money($command->amountCents, $command->currency);
                // On suppose que l'account est actif par défaut.
                $transaction = new DomainTransaction($command->transactionId, $command->clientId, $money, false);
            }

            // Soumission (idempotence garantie par l'utilisation du transactionId).
            $transaction->submit();

            // Calcul du score de risque : on utilise le profil récupéré en cache sinon la stratégie.
            if ($riskProfile !== null) {
                $riskScore = $riskProfile;
            } else {
                $riskScore = $this->riskScoringStrategy->calculateRiskScore($transaction);
            }
            $transaction->applyRiskScore($riskScore);

            // Sauvegarde de la transaction (optimistic locking dans l'implémentation réelle).
            $this->transactionRepository->save($transaction);

            // Publication des événements via le bus RabbitMQ (pattern outbox transactionnel).
            foreach ($transaction->getRecordedEvents() as $event) {
                $this->eventBus->publish($event);
            }
        } catch (Exception $e) {
            $span->end();
            throw $e;
        }

        $latency = ((microtime(true) - $startTime) * 1000); // en ms
        $this->logger->info('transaction.submitted', [
            'transaction_id' => $command->transactionId,
            'risk_score' => ($riskProfile !== null ? $riskProfile->value : null),
            'risk_level' => ($riskProfile !== null ? (string)$riskProfile->level() : null),
            'latency_ms' => $latency,
            'cache_hit' => $cacheHit,
            'trace_id' => $span->getContext()->getTraceId(),
        ]);
        $this->metrics->histogram('transaction_scoring_duration_ms', $latency, [
            'risk_level' => ($riskProfile !== null ? (string)$riskProfile->level() : null)
        ]);
        $span->end();
    }

    private function getTTLForRiskLevel(RiskLevel $level): int {
        switch ($level) {
            case RiskLevel::LOW:
                return 3600;
            case RiskLevel::MEDIUM:
                return 600;
            case RiskLevel::HIGH:
                return 60;
            case RiskLevel::CRITICAL:
                return 0; // aucun cache pour les cas critiques
            default:
                return 300;
        }
    }

    private array $outbox = [];

    public function publish(object $event): void {
        // Génération d'un event_id unique si nécessaire.
        if (!isset($event->eventId)) {
            $reflection = new \ReflectionClass($event);
            $eventName = $reflection->getShortName();
            $eventId = uniqid($eventName . '_', true);
            if (method_exists($event, 'setEventId')) {
                $event->setEventId($eventId);
            }
        }
        // Ajout dans l'outbox pour publication transactionnelle.
        $this->outbox[] = $event;
    }

    /**
     * Simule la flush de l'outbox (publication at-least-once).

    public function flush(): void {
        foreach ($this->outbox as $event) {
            // Dans une implémentation réelle, cette méthode enverrait l'événement via RabbitMQ.
            error_log("Publishing event: " . get_class($event));
        }
        $this->outbox = [];
    }

class MLRiskScoringStrategy implements RiskScoringStrategy {
    private HeuristicRiskScoringStrategy $fallback;
    private bool $circuitOpen = false;
    private int $failureCount = 0;

    public function __construct(HeuristicRiskScoringStrategy $fallback) {
        $this->fallback = $fallback;
    }

    /**
     * Simule une requête HTTP avec un timeout de 50ms maximum.
     */
    public function calculateRiskScore(Transaction $transaction): RiskScore {
        if ($this->circuitOpen) {
            return $this->fallback->calculateRiskScore($transaction);
        }
        try {
            // Ici, on simulerait une requête HTTP. Pour la démo, on retourne un score fixe.
            // Exemple : si le montant est élevé, on peut simuler un score critique (800).
            $money = $transaction->getMoney();
            // Pour la simulation, on utilise simplement la valeur en cents :
            $scoreValue = ($money->amount >= 1000) ? 800 : 200;
            $this->failureCount = 0; // Réinitialisation après succès.
            return new RiskScore($scoreValue);
        } catch (Exception $e) {
            $this->failureCount++;
            if ($this->failureCount >= 3) { // seuil pour ouvrir le circuit-breaker
                $this->circuitOpen = true;
            }
            throw new Exception("MLRiskScoringStrategy failed: " . $e->getMessage());
        }
    }
}

class TransactionController extends AbstractController {
    private SubmitTransactionCommandHandler $commandHandler;
    private TokenExtractorInterface $tokenExtractor;
    private SecurityContextInterface $securityContext;

    public function __construct(SubmitTransactionCommandHandler $commandHandler, TokenExtractorInterface $tokenExtractor, SecurityContextInterface $securityContext) {
        $this->commandHandler = $commandHandler;
        $this->tokenExtractor = $tokenExtractor;
        $this->securityContext = $securityContext;
    }

    /**
     * @Route("/api/v1/transactions", name="submit_transaction", methods={"POST"})
     */
    public function submitTransaction(Request $request): Response {
        // Vérification de la limitation de taux (simulée ici).
        if (!$this->isAllowed($request)) {
            return new Response(json_encode([
                "error" => "Rate limit exceeded"
            ]), 429, ['Content-Type' => 'application/json', 'Retry-After' => 60]);
        }
        
        // Extraction et validation du JWT RS256.
        $jwt = $this->tokenExtractor->extract($request);
        if (!$jwt) {
            return new Response(json_encode([
                "error" => "Unauthorized: Missing or invalid token"
            ]), 401, ['Content-Type' => 'application/json']);
        }
        
        try {
            $claims = $this->securityContext->validate($jwt);
        } catch (\Exception $e) {
            return new Response(json_encode([
                "error" => "Unauthorized: " . $e->getMessage()
            ]), 401, ['Content-Type' => 'application/json']);
        }
        
        if (!isset($claims['scope']) || strpos($claims['scope'], 'transactions:write') === false) {
            return new Response(json_encode([
                "error" => "Forbidden: Insufficient scope"
            ]), 403, ['Content-Type' => 'application/json']);
        }
        
        // Validation du payload via Symfony Validator (auto-wired).
        $data = json_decode($request->getContent(), true);
        if ($data === null || !is_array($data)) {
            return new Response(json_encode([
                "error" => "Bad Request: Invalid JSON"
            ]), 400, ['Content-Type' => 'application/json']);
        }
        
        // Limitation de la taille du payload (10KB max).
        if (strlen($request->getContent()) > 10240) {
            return new Response(json_encode([
                "error" => "Payload Too Large"
            ]), 413, ['Content-Type' => 'application/json']);
        }
        
        // Validation des champs requis.
        if (!isset($data['clientId'], $data['transactionId'], $data['amountCents'], $data['currency'])) {
            return new Response(json_encode([
                "error" => "Bad Request: Missing required fields"
            ]), 400, ['Content-Type' => 'application/json']);
        }
        
        // Protection contre injection SQL (échappement des entrées).
        $clientId = htmlspecialchars($data['clientId'], ENT_QUOTES);
        $transactionId = htmlspecialchars($data['transactionId'], ENT_QUOTES);
        $amountCents = intval($data['amountCents']);
        $currency = htmlspecialchars($data['currency'], ENT_QUOTES);
        
        // Création de la commande.
        $command = new SubmitTransactionCommand($clientId, $transactionId, $amountCents, $currency);
        
        try {
            $this->commandHandler->handle($command);
        } catch (\Exception $e) {
            return new Response(json_encode([
                "error" => "Bad Request: " . $e->getMessage()
            ]), 400, ['Content-Type' => 'application/json']);
        }
        
        // Réponse HTTP 202 Accepted avec en-têtes de sécurité.
        $response = new Response('', Response::HTTP_ACCEPTED);
        $response->headers->set('Strict-Transport-Security', 'max-age=31536000; includeSubDomains');
        $response->headers->set('Content-Security-Policy', "default-src 'self'");
        $response->headers->set('X-Frame-Options', 'DENY');
        
        return $response;
    }
