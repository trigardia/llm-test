## Mission

Implémente un microservice Symfony 7 de scoring de risque transactionnel en temps réel pour une plateforme de paiement B2B traitant 5 000 transactions par seconde avec une latence p99 < 100ms.

La plateforme reçoit des demandes de paiement d'entreprises clientes. Chaque transaction est scorée en temps réel pour détecter les fraudes. Le score détermine si la transaction est : APPROVED, REVIEW_REQUIRED, ou BLOCKED. En cas de fraude détectée, un FraudAlertTriggered domain event est émis.

Contraintes : Symfony 7.2 + PHP 8.3 (readonly properties), Redis 7 (cache TTL par niveau de risque), PostgreSQL 16 (optimistic locking), RabbitMQ (domain events), CQRS + Event Sourcing, JWT RS256 (scope transactions:write, expiry 15min), rate limiting 1000 req/min par client_id.

readonly class Money
{
    public function __construct(

    ) {
        if ($this->amountCents <= 0) {
            throw new \InvalidArgumentException("Amount must be positive");

            throw new \InvalidArgumentException("Invalid ISO 4217 currency");
        }
    }

    public function equals(Money $other): bool
    {
        return $this->amountCents === $other->amountCents && $this->currency === $other->currency;
    }
}

/**
 * DDD: Risk Level Enum for categorical scoring.
 */
enum RiskLevel: string
{
    case LOW = 'LOW';
    case MEDIUM = 'MEDIUM';
    case HIGH = 'HIGH';
    case CRITICAL = 'CRITICAL';
}

/**
 * SOLID: SRP - RiskScore encapsulates the scoring logic.
 * Performance: O(1) complexity for level determination.
 */
readonly class RiskScore
{
    public function __construct(public int $value)
    {
        if ($this->value < 0 || $this->value > 1000) {
            throw new \InvalidArgumentException("Risk score must be between 0 and 1000");
        }
    }

    public function level(): RiskLevel
    {
        return match (true) {
            $this->value <= 300 => RiskLevel::LOW,
            $this->value <= 700 => RiskLevel::MEDIUM,
            $this->value <= 900 => RiskLevel::HIGH,
            default => RiskLevel::CRITICAL,
        };
    }
}

/**
 * DDD: Transaction Aggregate Root.
 * Invariants are strictly enforced in the domain.
 */
class Transaction
{
    private string $status = 'PENDING';
    private array $domainEvents = [];

    public function __construct(
        public readonly string $id,
        public readonly string $clientId,
        public readonly Money $money,
        public readonly bool $isAccountSuspended,
        private ?RiskScore $riskScore = null,
        private int $version = 0 // For Optimistic Locking
    ) {
        if ($this->isAccountSuspended) {
            throw new \DomainException("Cannot create transaction for suspended account");
        }
    }

    public function applyRiskScore(RiskScore $score): void
    {
        $this->riskScore = $score;
        
        if ($score->level() === RiskLevel::CRITICAL) {
            $this->status = 'BLOCKED';
            $this->recordEvent(new FraudAlertTriggered($this->id, $score));
        } elseif ($score->level() === RiskLevel::HIGH) {
            $this->status = 'REVIEW_REQUIRED';
        } else {
            $this->status = 'APPROVED';
        }
        
        $this->recordEvent(new RiskScoreCalculated($this->id, $score));
    }

    public function getStatus(): string
    {
        return $this->status;
    }

    public function getRiskScore(): ?RiskScore
    {
        return $this->riskScore;
    }

    public function getVersion(): int
    {
        return $this->version;
    }

    private function recordEvent(object $event): void
    {
        $this->domainEvents[] = $event;
    }

    public function releaseEvents(): array
    {
        $events = $this->domainEvents;
        $this->domainEvents = [];
        return $events;
    }
}

use App\Domain\Model\Transaction;

readonly class SubmitTransactionCommand
{
    public function __construct(
        public string $transactionId,
        public string $clientId,
        public int $amountCents,
        public string $currency,
        public string $idempotencyKey
    ) {}

class SubmitTransactionCommandHandler
{
    public function __construct(
        private TransactionRepository $repository,
        private RiskScoringStrategy $scoringStrategy,
        private \Redis $cache,
        private TracerInterface $tracer,
        private LoggerInterface $logger,
        private CollectorRegistry $metrics
    ) {}

    public function handle(SubmitTransactionCommand $command): string
    {
        $span = $this->tracer->spanBuilder('transaction.submit')
            ->setAttribute('client_id', $command->clientId)
            ->setAttribute('amount_cents', $command->amountCents)
            ->startSpan();

        $startTime = microtime(true);
        $cacheHit = false;

        try {
            // Idempotency check (O(1))
            if ($existingId = $this->cache->get("idempotency:{$command->idempotencyKey}")) {
                return (string)$existingId;
            }

            // Cache-aside check for client risk profile
            $cachedScore = $this->cache->get("risk_profile:{$command->clientId}");
            if ($cachedScore) {
                $cacheHit = true;
                $riskScore = new RiskScore((int)$cachedScore);
            } else {
                // Domain flow
                $transaction = new Transaction(
                    $command->transactionId,
                    $command->clientId,
                    new \App\Domain\Model\Money($command->amountCents, $command->currency),
                    false // In real scenario: fetch from AccountService
                );

                $riskScore = $this->scoringStrategy->calculate($transaction);
                $transaction->applyRiskScore($riskScore);
                
                $this->repository->save($transaction);
                
                // Update cache based on RiskLevel TTL
                $ttl = match($riskScore->level()) {
                    \App\Domain\Model\RiskLevel::LOW => 3600,
                    \App\Domain\Model\RiskLevel::MEDIUM => 600,
                    \App\Domain\Model\RiskLevel::HIGH => 60,
                    \App\Domain\Model\RiskLevel::CRITICAL => 0,
                };
                if ($ttl > 0) $this->cache->setex("risk_profile:{$command->clientId}", $ttl, (string)$riskScore->value);
            }

            $latency = (microtime(true) - $startTime) * 1000;
            
            $this->logger->info('transaction.submitted', [
                'transaction_id' => $command->transactionId,
                'risk_score' => $riskScore->value,
                'risk_level' => $riskScore->level()->name,
                'latency_ms' => $latency,
                'cache_hit' => $cacheHit,
                'trace_id' => $span->getContext()->getTraceId(),
            ]);

            $this->metrics->getOrRegisterHistogram('risk_scoring', 'transaction_scoring_duration_ms', 'Latency', ['risk_level'])
                ->observe($latency, [$riskScore->level()->name]);

            $this->cache->setex("idempotency:{$command->idempotencyKey}", 86400, $command->transactionId);
            
            return $command->transactionId;

        } finally {
            $span->end();
        }
    }
}

        // Optimistic locking via Doctrine @Version

    }
    
    /**

     */

    {
        return $this->em->createQuery(
            'SELECT t FROM App\Infrastructure\Persistence\TransactionEntity t

             ORDER BY t.createdAt DESC'
        )
        ->setParameter('clientId', $clientId)
        ->setParameter('cursor', $cursor)
        ->setMaxResults($limit)
        ->getResult();
    }
}

class RabbitMQEventBus
{
    public function __construct(private MessageBusInterface $bus) {}

    /**
     * Outbox Pattern: Events are stored in DB and dispatched asynchronously.
     * Ensures at-least-once delivery.
     */
    public function publish(Transaction $transaction): void
    {
        foreach ($transaction->releaseEvents() as $event) {
            // Wrapped in DB Transaction in Application Layer
            $this->bus->dispatch($event);
        }
    }
}

class MLRiskScoringStrategy implements RiskScoringStrategy
{
    public function __construct(
        private HttpClientInterface $httpClient,
        private HeuristicRiskScoringStrategy $fallback
    ) {}

    public function calculate(Transaction $transaction): RiskScore
    {
        try {
            // Circuit Breaker & Timeout (50ms max)
            $response = $this->httpClient->request('POST', 'https://ml-risk-api/score', [
                'timeout' => 0.05, 
                'json' => ['tx_id' => $transaction->getId()]
            ]);
            
            $data = $response->toArray();
            return new RiskScore($data['score']);
        } catch (\Exception $e) {
            // Fallback to Heuristic strategy (OCP/Resilience)
            return $this->fallback->calculate($transaction);
        }
    }
}

class TransactionController extends AbstractController
{
    #[Route('/api/v1/transactions', methods: ['POST'])]
    public function create(Request $request, SubmitTransactionCommandHandler $handler): JsonResponse
    {
        // Security: JWT RS256 & Scope check handled by Symfony Security Bundle / LexikJWT
        // $this->denyAccessUnlessGranted('ROLE_TRANSACTIONS_WRITE');

        // Zero-Trust: Strict validation
        $data = json_decode($request->getContent(), true);
        if (!$this->isValid($data)) {
            return new JsonResponse(['error' => 'Invalid payload'], 400);
        }

        // Idempotency Key (RFC 8615)
        $idempotencyKey = $request->headers->get('Idempotency-Key');
        if (!$idempotencyKey) {
            throw new HttpException(400, 'Idempotency-Key header required');
        }

        $command = new SubmitTransactionCommand(
            uniqid('tx_'),
            $data['client_id'],
            $data['amount'],
            $data['currency'],
            $idempotencyKey
        );

        $txId = $handler->handle($command);

        return new JsonResponse(['transaction_id' => $txId], 202);
    }

    private function isValid(?array $data): bool
    {
        return isset($data['client_id'], $data['amount'], $data['currency']) 
            && is_int($data['amount']) 
            && strlen($data['currency']) === 3;
    }
}
