## Mission

Implémente un microservice Symfony 7 de scoring de risque transactionnel en temps réel pour une plateforme de paiement B2B traitant 5 000 transactions par seconde avec une latence p99 < 100ms.

La plateforme reçoit des demandes de paiement d'entreprises clientes. Chaque transaction est scorée en temps réel pour détecter les fraudes. Le score détermine si la transaction est : APPROVED, REVIEW_REQUIRED, ou BLOCKED. En cas de fraude détectée, un FraudAlertTriggered domain event est émis.

namespace App\Domain\Transaction;

use App\Domain\Transaction\ValueObject\Money;

use App\Domain\Transaction\Exception\InvalidMoneyException;
use App\Domain\Transaction\Exception\AccountSuspendedException;
use App\Domain\Transaction\Exception\InvalidRiskScoreException;

final class Transaction
{
    private string $id;
    private string $clientId;
    private Money $amount;
    private string $currency;
    private string $status;
    private ?RiskScore $riskScore = null;
    private \DateTimeImmutable $createdAt;
    private \DateTimeImmutable $updatedAt;
    private bool $isAccountSuspended;

    public function __construct(
        string $id,
        string $clientId,
        Money $amount,
        string $currency,
        bool $isAccountSuspended
    ) {
        $this->id = $id;
        $this->clientId = $clientId;
        $this->amount = $amount;
        $this->currency = $currency;
        $this->status = 'PENDING';
        $this->createdAt = new \DateTimeImmutable();
        $this->updatedAt = $this->createdAt;
        $this->isAccountSuspended = $isAccountSuspended;

        $this->ensureValidState();
    }

    public function submit(): void
    {
        if ($this->isAccountSuspended) {
            throw new AccountSuspendedException("Account {$this->clientId} is suspended");
        }

        if ($this->amount->getAmount() <= 0) {
            throw new InvalidMoneyException("Amount must be positive");
        }

        $this->status = 'PENDING';
        $this->updatedAt = new \DateTimeImmutable();
        $this->ensureValidState();
    }

    public function calculateRiskScore(RiskScore $riskScore): void
    {
        if ($this->status !== 'PENDING') {
            throw new InvalidTransactionStateException("Cannot calculate risk score for non-pending transaction");
        }

        $this->riskScore = $riskScore;
        $this->updatedAt = new \DateTimeImmutable();

        if ($riskScore->level() === RiskLevel::CRITICAL) {
            $this->status = 'BLOCKED';
        } elseif ($riskScore->level() === RiskLevel::HIGH) {
            $this->status = 'REVIEW_REQUIRED';
        } else {
            $this->status = 'APPROVED';
        }

        $this->ensureValidState();
    }

    public function getId(): string
    {
        return $this->id;
    }

    public function getClientId(): string
    {
        return $this->clientId;
    }

    public function getAmount(): Money
    {
        return $this->amount;
    }

    public function getStatus(): string
    {
        return $this->status;
    }

    public function getRiskScore(): ?RiskScore
    {
        return $this->riskScore;
    }

    public function getCreatedAt(): \DateTimeImmutable
    {
        return $this->createdAt;
    }

    public function getUpdatedAt(): \DateTimeImmutable
    {
        return $this->updatedAt;
    }

    private function ensureValidState(): void
    {
        if ($this->amount->getAmount() <= 0) {
            throw new InvalidMoneyException("Amount must be positive");
        }

        if ($this->isAccountSuspended && $this->status !== 'BLOCKED') {
            throw new AccountSuspendedException("Suspended account cannot have non-blocked status");
        }

        if (!in_array($this->status, ['PENDING', 'APPROVED', 'REVIEW_REQUIRED', 'BLOCKED'])) {
            throw new InvalidTransactionStateException("Invalid transaction status");
        }

        if ($this->riskScore !== null && $this->riskScore->value() < 0 || $this->riskScore->value() > 1000) {
            throw new InvalidRiskScoreException("Risk score must be between 0 and 1000");
        }
    }

    public function pullDomainEvents(): array
    {
        $events = [];

        if ($this->status === 'PENDING') {
            $events[] = new TransactionSubmitted($this->id, $this->clientId, $this->amount, $this->currency);
        }

        if ($this->riskScore !== null) {
            $events[] = new RiskScoreCalculated($this->id, $this->riskScore->value(), $this->riskScore->level()->name);

            if ($this->riskScore->level() === RiskLevel::CRITICAL) {
                $events[] = new FraudAlertTriggered($this->id, $this->clientId, $this->amount->getAmount());
            }
        }

        return $events;
    }
}

final class SubmitTransactionCommandHandler
{
    public function __construct(
        private TransactionRepository $repository,
        private RiskScoringStrategy $scoringStrategy,
        private EventBus $eventBus,
        private RedisRiskProfileCache $cache,
        private TracerInterface $tracer,
        private LoggerInterface $logger,
        private MetricFactory $metrics
    ) {}

    public function __invoke(SubmitTransactionCommand $command): void
    {
        $span = $this->tracer->spanBuilder('transaction.submit')
            ->setAttribute('client_id', $command->clientId)
            ->setAttribute('amount_cents', $command->amountCents)
            ->startSpan();

        $startTime = microtime(true);

        try {
            // Check cache first (Zero-Trust: validate cache response)
            $cachedScore = $this->cache->get($command->clientId);
            $cacheHit = $cachedScore !== null;

            $transaction = $this->repository->findById($command->transactionId);

            if ($transaction === null) {
                $transaction = new Transaction(
                    $command->transactionId,
                    $command->clientId,
                    new Money($command->amountCents, $command->currency),
                    $command->currency,
                    false // Account suspension would be checked in real implementation
                );
            }

            $transaction->submit();

            // Calculate risk score
            $riskScore = $this->scoringStrategy->calculateRiskScore($transaction);
            $transaction->calculateRiskScore($riskScore);

            $this->repository->save($transaction);

            // Publish domain events
            foreach ($transaction->pullDomainEvents() as $event) {
                $this->eventBus->publish($event);
            }

            // Update cache with TTL based on risk level
            $this->cache->set(
                $command->clientId,
                $riskScore->value(),
                $this->getCacheTTL($riskScore->level())
            );

            $latency = (microtime(true) - $startTime) * 1000;

            $this->logger->info('transaction.submitted', [
                'transaction_id' => $command->transactionId,
                'risk_score' => $riskScore->value(),
                'risk_level' => $riskScore->level()->name,
                'latency_ms' => $latency,
                'cache_hit' => $cacheHit,
                'trace_id' => $span->getContext()->getTraceId(),
            ]);

            $this->metrics->histogram('transaction_scoring_duration_ms', $latency,
                ['risk_level' => $riskScore->level()->name]);

        } finally {
            $span->end();
        }
    }

    private function getCacheTTL(RiskLevel $level): int
    {
        return match ($level) {
            RiskLevel::LOW => 3600,
            RiskLevel::MEDIUM => 600,
            RiskLevel::HIGH => 60,
            RiskLevel::CRITICAL => 0,
        };
    }
}

final class GetRiskReportQueryHandler
{
    public function __construct(
        private TransactionRepository $repository,
        private RedisRiskProfileCache $cache,
        private TracerInterface $tracer,
        private LoggerInterface $logger
    ) {}

    public function __invoke(GetRiskReportQuery $query): array
    {
        $span = $this->tracer->spanBuilder('risk_report.query')
            ->setAttribute('client_id', $query->clientId)
            ->startSpan();

        try {
            // First check cache
            $cachedReport = $this->cache->getReport($query->clientId);
            if ($cachedReport !== null) {
                $this->logger->info('risk_report.cache_hit', [
                    'client_id' => $query->clientId,
                    'trace_id' => $span->getContext()->getTraceId(),
                ]);
                return $cachedReport;
            }

            // Fallback to database
            $transactions = $this->repository->findByClientId($query->clientId);

            // Filter by date range if provided
            $filtered = array_filter($transactions, function(Transaction $t) use ($query) {
                if ($query->startDate !== null && $t->getCreatedAt() < $query->startDate) {
                    return false;
                }
                if ($query->endDate !== null && $t->getCreatedAt() > $query->endDate) {
                    return false;
                }
                return true;
            });

            $report = array_map(function(Transaction $t) {
                return [
                    'transaction_id' => $t->getId(),
                    'amount' => $t->getAmount()->getAmount(),
                    'currency' => $t->getAmount()->getCurrency(),
                    'status' => $t->getStatus(),
                    'risk_score' => $t->getRiskScore()?->value(),
                    'risk_level' => $t->getRiskScore()?->level()->name,
                    'created_at' => $t->getCreatedAt()->format('c'),
                ];
            }, $filtered);

            // Cache the report with 300s TTL
            $this->cache->setReport($query->clientId, $report, 300);

            $this->logger->info('risk_report.generated', [
                'client_id' => $query->clientId,
                'count' => count($report),
                'trace_id' => $span->getContext()->getTraceId(),
            ]);

            return $report;
        } finally {
            $span->end();
        }
    }
}

final class DoctrineTransactionRepository implements TransactionRepository
{
    public function __construct(
        private EntityManagerInterface $em,
        private Connection $connection
    ) {}

    public function findById(string $id): ?Transaction
    {
        return $this->em->find(TransactionEntity::class, $id);
    }

    public function save(Transaction $transaction): void
    {
        $entity = $this->em->find(TransactionEntity::class, $transaction->getId());

        if ($entity === null) {
            $entity = new TransactionEntity();
            $entity->setId($transaction->getId());
        }

        $entity->setClientId($transaction->getClientId());
        $entity->setAmount($transaction->getAmount()->getAmount());
        $entity->setCurrency($transaction->getAmount()->getCurrency());
        $entity->setStatus($transaction->getStatus());
        $entity->setRiskScore($transaction->getRiskScore()?->value());
        $entity->setRiskLevel($transaction->getRiskScore()?->level()->name);
        $entity->setCreatedAt($transaction->getCreatedAt());
        $entity->setUpdatedAt($transaction->getUpdatedAt());

        $this->em->persist($entity);
        $this->em->flush();
    }

    public function findByClientId(string $clientId, int $limit = 100): array
    {
        $query = $this->connection->createQueryBuilder()
            ->select('t')
            ->from(TransactionEntity::class, 't')
            ->where('t.clientId = :clientId')
            ->setParameter('clientId', $clientId)
            ->orderBy('t.createdAt', 'DESC')
            ->setMaxResults($limit)
            ->getQuery();

        return $query->getResult();
    }
}

namespace App\Infrastructure\Scoring;

final class MLRiskScoringStrategy implements RiskScoringStrategy
{
    public function __construct(
        private HttpClientInterface $httpClient,
        private string $mlServiceUrl,
        private HeuristicRiskScoringStrategy $fallbackStrategy,
        private TracerInterface $tracer,
        private LoggerInterface $logger
    ) {}

    public function calculateRiskScore(Transaction $transaction): RiskScore
    {
        $span = $this->tracer->spanBuilder('ml_scoring')
            ->setAttribute('client_id', $transaction->getClientId())
            ->setAttribute('amount', $transaction->getAmount()->getAmount())
            ->startSpan();

        try {
            $startTime = microtime(true);

            try {
                $response = $this->httpClient->request('POST', $this->mlServiceUrl, [
                    'json' => [
                        'client_id' => $transaction->getClientId(),
                        'amount' => $transaction->getAmount()->getAmount(),
                        'currency' => $transaction->getAmount()->getCurrency(),
                    ],
                    'headers' => ['Accept' => 'application/json'],
                    'timeout' => 0.05, // 50ms
                ]);

                $data = json_decode($response->getContent(), true);

                if (isset($data['score'])) {
                    $latency = (microtime(true) - $startTime) * 1000;
                    $this->logger->info('ml_scoring.success', [
                        'score' => $data['score'],
                        'latency_ms' => $latency,
                        'trace_id' => $span->getContext()->getTraceId(),
                    ]);
                    return new RiskScore($data['score']);
                }
            } catch (\Exception $e) {
                $this->logger->warning('ml_scoring.fallback', [
                    'error' => $e->getMessage(),
                    'trace_id' => $span->getContext()->getTraceId(),
                ]);
            }

            // Fallback to heuristic strategy
            return $this->fallbackStrategy->calculateRiskScore($transaction);
        } finally {
            $span->end();
        }
    }
}

final class HeuristicRiskScoringStrategy implements RiskScoringStrategy
{
    public function calculateRiskScore(Transaction $transaction): RiskScore
    {
        // Simple heuristic: higher amounts = higher risk
        $amount = $transaction->getAmount()->getAmount();
        $score = min(1000, max(0, (int)(log($amount) * 100)));

        return new RiskScore($score);
    }
}

                return new Response('Payload too large', Response::HTTP_PAYLOAD_TOO_LARGE);
            }
